tradition_rezzou_raids = {
	category = combat

	layers = {
		0 = martial
		1 = mena
		4 = rider.dds
	}

	is_shown = {
		has_cultural_pillar = heritage_imazighen
	}
	can_pick = {
		has_cultural_pillar = heritage_imazighen
	}

	parameters = {
		unlocks_bilad_rezzou_raid_intent = yes
		prowess_traits_more_valued = yes
	}

	character_modifier = {
		prowess = 1
		knight_effectiveness_mult = 0.1
		light_cavalry_damage_mult = 0.1
		camel_cavalry_damage_mult = 0.1
		raid_speed = 0.1
	}

	cost = {
		prestige = {
			add = {
				value = tradition_base_cost
				desc = BASE
				format = "BASE_VALUE_FORMAT"
			}
			if = {
				limit = {
					NOT = {
						culture_pillar:ethos_bellicose = { is_in_list = traits }
					}
				}
				add = {
					value = tradition_incompatible_ethos_penalty
					desc = not_bellicose_desc
				}
			}
			multiply = tradition_replacement_cost_if_relevant
		}
	}

	ai_will_do = {
		value = 150
	}
}

# Yusseter:
# BILAD does not currently depend on RICE, so the RICE Kel Tagelmust key is imported locally.
# The key stays tradition_RICE_tuareg for compatibility with doc hooks; RICE-only decision/activity unlocks are intentionally omitted.
tradition_RICE_tuareg = {
	category = regional

	layers = {
		0 = diplo
		1 = mena
		4 = instrument.dds
	}

	is_shown = {
		OR = {
			this = culture:touareg
			any_parent_culture_or_above = {
				this = culture:touareg
			}
		}
		any_culture_county = {
			any_county_province = {
				geographical_region = world_africa
			}
		}
	}

	can_pick = {
		OR = {
			this = culture:touareg
			any_parent_culture_or_above = {
				this = culture:touareg
			}
		}
		custom_tooltip = {
			text = BILAD_tuareg_african_counties_desc
			any_culture_county = {
				count >= 5
				any_county_province = {
					geographical_region = world_africa
				}
			}
		}
	}

	parameters = {
		unlocks_bilad_ahal_gathering = yes
		number_of_spouses = 1
		gender_equal_inheritance = yes
	}

	character_modifier = {
		monthly_prestige_gain_per_happy_powerful_vassal_mult = 0.03
	}

	cost = {
		prestige = {
			add = {
				value = tradition_base_cost
				desc = BASE
				format = "BASE_VALUE_FORMAT"
			}
			if = {
				limit = {
					NOR = {
						culture_pillar:ethos_communal = { is_in_list = traits }
						culture_pillar:ethos_egalitarian = { is_in_list = traits }
						culture_pillar:ethos_stoic = { is_in_list = traits }
					}
				}
				add = {
					value = tradition_incompatible_ethos_penalty
					desc = not_communal_egalitarian_or_stoic_desc
				}
			}
			multiply = tradition_replacement_cost_if_relevant
		}
	}

	ai_will_do = {
		value = 100
		if = {
			limit = {
				culture_tradition:tradition_monogamous = { is_in_list = traits }
			}
			multiply = 0.25
		}
		if = {
			limit = {
				culture_tradition:tradition_equal_inheritance = { is_in_list = traits }
			}
			multiply = 0.25
		}
	}
}
