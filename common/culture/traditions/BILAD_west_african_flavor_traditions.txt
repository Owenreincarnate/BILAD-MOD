tradition_african_brewery = {
	category = ritual

	layers = {
		0 = learning
		1 = mena
		4 = crown.dds
	}

	is_shown = {
		OR = {
			has_cultural_pillar = heritage_west_african
			has_cultural_pillar = heritage_senegambian
		}
	}

	can_pick = {
		OR = {
			has_cultural_pillar = heritage_west_african
			has_cultural_pillar = heritage_senegambian
		}
	}

	parameters = {
		allows_bilad_millet_beer_pubs = yes
		visit_millet_beer_pubs_active = yes
	}

	character_modifier = {
		stress_loss_mult = 0.05
		same_culture_opinion = 5
	}

	cost = {
		prestige = {
			add = tradition_base_cost
			if = {
				limit = { has_cultural_pillar = heritage_west_african }
				multiply = 0.75
			}
			if = {
				limit = { has_cultural_pillar = heritage_senegambian }
				multiply = 0.75
			}
		}
	}

	ai_will_do = {
		value = 80
	}
}
