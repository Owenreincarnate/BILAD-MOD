# The central Saharan rainy season is represented as June through September.
# Keeping it in one trigger makes the balance window easy to adjust later.
bilad_rezzou_season_trigger = {
	current_month >= 6
	current_month <= 9
}

bilad_can_use_rezzou_intent_trigger = {
	culture = {
		has_cultural_parameter = unlocks_bilad_rezzou_raid_intent
	}
}
