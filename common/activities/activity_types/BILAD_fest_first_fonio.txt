festival_first_fonio = {
    is_shown = {
        has_cultural_tradition = tradition_fulbe_pastorialism
        highest_held_title_tier > tier_barony
    }

    can_start_showing_failures_only = {
        NOT = {is_activity_type_on_cooldown = festival_first_fonio}
        is_available_adult = yes
    }

    is_valid = {
		scope:host = {
			is_alive = yes
			is_capable_adult = yes
			is_imprisoned = no
			is_landed_or_landless_administrative = yes
			NOT = { is_incapable = yes }
		}        
    }

	on_enter_active_state = {
		surveyor_no_stone_unturned_perk_effect = yes
		if = {
			limit = {
				this = scope:host
				has_perk = personal_touch_perk
			}
			location.county = {
				local_adjudicate_effect = yes
			}
		}
	}

	on_host_death = {
		invalidation_host_death_effect = yes
	}

	on_invalidated = {
		invalidation_imprisoned_effect = yes
		invalidation_incapable_effect = yes
		invalidation_unlanded_effect = yes
	}

    # Parameters

    is_single_location = yes

    province_filter = domicile_domain
    ai_province_filter = domicile_domain

    max_route_deviation_mult = 2.0

    is_location_valid = {
        has_holding = yes
        # Only available in regions where Fonio was historically distributed, update once geographical_region files added
        # geographical_region = 
    }

    cooldown = {years = 5}

    ai_will_do = {

    }

	# Guests - only from court, smaller scale activity

	allow_zero_guest_invites = yes
	open_invite = no
    max_guests = 20

	can_be_activity_guest = {
		is_ai = yes
		court_owner ?= scope:host
		is_healthy = yes
		is_landed = no
		age >= 12
		OR = {
			is_knight_of = scope:host
			is_councillor_of = scope:host
			is_child_of = scope:host
			is_heir_of = scope:host
			is_spouse_of = scope:host
			is_consort_of = scope:host

	}

		base = 10

		# Up to how many characters to select for an AI

	# Costs

    cost = {

	# Phases

	phases = {
		harvest_festival = {
			is_predefined = yes

			on_phase_active = {
				if = {
					limit = {this = scope:host}
					scope:activity = { progress_activity_phase_after = { weeks = 3 } }
				}
				scope:host = {
					trigger_event = fest_fonio.0001
				}
			}

			on_weekly_pulse = {
				# Fires for every attendee
				root.culture = {
					if = {
						limit = {
							NOT = {
								has_variable = fonio_fest_week
							}
						}
						set_variable = {
							name = fonio_fest_week
							value = 1
						}
					}
					else = {
						change_variable = {
							name = fonio_fest_week
							add = 1
						}
					}
				}

				if = {
					limit = {scope:activity.var:week_flag = 1}
					scope:host = {
						trigger_event = {
							fest_fonio.0002
						}
					}
				}

				else_if = {
					limit = {scope:activity.var:week_flag = 2}
					scope:host = {
						trigger_event = {
							fest_fonio.0003
						}
					}
				}

				else_if = {
					limit = {scope:activity.var:week_flag = 3}
					scope:host = {
						trigger_event = {
							fest_fonio.0004
						}
					}
				}
			}

			on_end = {
				scope:host = {
					trigger_event = {
						fest_fonio.0005
					}
				}
			}
		}
	}
