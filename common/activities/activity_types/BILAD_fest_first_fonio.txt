        has_cultural_tradition = tradition_fulbe_pastorialism

        NOT = {is_activity_type_on_cooldown = festival_first_fonio}

    is_valid = {
		scope:host = {
			is_alive = yes
			is_capable_adult = yes

		surveyor_no_stone_unturned_perk_effect = yes

			limit = {

				local_adjudicate_effect = yes
			}
		}
	}

	on_host_death = {
		invalidation_host_death_effect = yes

	on_invalidated = {
		invalidation_imprisoned_effect = yes
		invalidation_incapable_effect = yes

    # Parameters

    is_single_location = yes

    province_filter = domicile_domain
    ai_province_filter = domicile_domain

    max_route_deviation_mult = 2.0

    is_location_valid = {
        has_holding = yes
        # Only available in regions where Fonio was historically distributed, update once geographical_region files added
        # geographical_region =

    ai_will_do = {

    }

	# Phases

	phases = {
		harvest_festival = {
			is_predefined = yes

			on_phase_active = {
				if = {
					limit = {this = scope:host}
					scope:activity = { progress_activity_phase_after = { weeks = 3 } }
					scope:host = {
						trigger_event = fest_fonio.0001
					}
				}
			}

			on_weekly_pulse = {
				# Fires for every attendee
				if = {
					limit = {
						this = scope:host
					}
					scope:activity = {
						if = {
							limit = {
								NOT = {
									has_variable = fonio_fest_week
								}
							}
							set_variable = {
								name = fonio_fest_week
								value = 1
							}
						}
						else = {
							change_variable = {
								name = fonio_fest_week
								add = 1
							}
						}
					}
				}
					
				if = {
					limit = {
						scope:activity.var:fonio_fest_week = 1
						this = scope:host
					}
					scope:host = {
						trigger_event = fest_fonio.0002
					}
				}

				else_if = {
					limit = {
						scope:activity.var:fonio_fest_week = 2
						this = scope:host
					}
					scope:host = {
						trigger_event = fest_fonio.0003
					}
				}

				else_if = {
					limit = {
						scope:activity.var:fonio_fest_week = 3
						this = scope:host
					}
					scope:host = {
						trigger_event = fest_fonio.0004
					}
				}
			}

			on_end = {
				if = {
					limit = {
						this = scope:host
					}
					scope:host = {
						trigger_event = fest_fonio.0005
					}
				}
			}
		}
	}
