ahal_gathering = {
	activity_group_type = activities
	sort_order = 25
	# Unlocked by the local Kel Tagelmust cultural tradition.
	is_shown = {
		bilad_has_kel_tagelmust_trigger = yes
		highest_held_title_tier > tier_barony
	}

	can_start_showing_failures_only = {
		NOT = { is_activity_type_on_cooldown = ahal_gathering }
		is_available_adult = yes
		is_imprisoned = no
	}

	is_valid = {
		scope:host = {
			is_alive = yes
			is_capable_adult = yes
			is_imprisoned = no
			is_landed = yes
		}
	}

	on_host_death = {
		invalidation_host_death_effect = yes

	province_filter = domicile_domain
	ai_province_filter = domicile_domain
	max_province_icons = 8

	is_location_valid = {
		OR = {
			has_holding = yes
			scope:host.domicile.domicile_location ?= this
		}
		OR = {
			terrain = desert
			terrain = drylands
			terrain = oasis
			terrain = desert_mountains
		}
	}

	province_score = {
		value = 10
		if = {
			limit = { terrain = oasis }

		}

			limit = { this = scope:host.capital_province }

		}

		base = 35
		modifier = {
			add = 25

		modifier = {
			add = 20
			has_relation_friend = scope:host
		}
	}

	host_intents = {
		intents = { reduce_stress_intent }

		intents = { reduce_stress_intent }

			order = 1
